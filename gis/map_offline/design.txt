# 原始设计
  图层名                            zoom        说明                                            
  -----------------------------------------------------------------------------------------------
  boundary_admin_country            z0–z6       国家级边界。                                    
  boundary_admin_state              z3–z10      省/州级边界。                                   
  boundary_admin_subdivision        z6–z12      市/县/区级边界。                                
  places_country_state              z0–z6       国家/州/省级点或面中心点。                      
  places_city                       z4–z12      城市级地点。                                    
  places_town_village               z7–z14      乡镇/村级地点。                                 
  places_local                      z10–z14     更小的居住点、社区、街区。                      
  places_a                          z7–z12      城市面状边界，仅在需要“城市面”视觉时保留。    
  roads_motorway                    z6–z14      高速/快速路。                                   
  roads_primary_secondary           z8–z14      主干/次干道路。                                 
  roads_minor                       z10–z14     支路、服务路、一般小路。                        
  roads_path                        z12–z14     步道、自行车道、非机动车小路。                  
  railways                          z8–z14      铁路。                                          
  waterways_river                   z8–z14      主河流。                                        
  waterways_canal                   z9–z14      运河/人工水道。                                 
  waterways_stream                  z10–z14     支流、小溪、排水沟。                            
  water                             z4–z14      湖泊、海洋、宽水体。                            
  natural_a_forest                  z4–z14      森林、植被面。                                  
  natural_a_mountain                z6–z14      山地、冰川、裸地等大面。                        
  natural_peak                      z8–z14      山峰点。                                        
  natural_tree                      z12–z14     树点/小型自然点要素。                           
  landuse_residential_commercial    z8–z14      居住、商业、城市建设用地。                      
  landuse_industrial                z8–z14      工业用地。                                      
  landuse_recreation                z8–z14      公园、体育、休闲设施用地。                      
  landuse_agriculture               z8–z14      农业、牧场等。                                  
  protected                         z7–z12      保护区。                                        
  pois_major                        z10–z14     重要 POI。                                      
  pois_minor                        z12–z14     一般 POI。                                      
  pofw                              z10–z14     宗教场所。                                      
  transport_station                 z10–z14     公交站、地铁站、交通点。                        
  traffic_signal                    z12–z14     信号灯。                                        
  buildings_major                   z13–z14     地标建筑、重要建筑。                            
  buildings_minor                   z13–z14     普通建筑面。
